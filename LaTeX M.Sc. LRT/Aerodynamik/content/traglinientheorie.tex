                \centering
                Nachlauf-Zirkulationsfeld:\\ \vspace{-5mm}
                \begin{align*}
                    \gamma(y) = \frac{\partial\Gamma(y)}{\partial y}
                \end{align*}
                Kutta-Jukowski:\\ \vspace{-5mm}
                \begin{align*}
                    \text{d}A = \rho\,U_\infty\,\Gamma(y)\,\text{d}y = C_\text{A}(y)\,\frac{\rho\,U_\infty^2}{2}\,t(y)\,\text{d}y
                \end{align*}
            \end{minipage}%
            \begin{minipage}{0.5\cellwidth-\TileSep}
                \centering
                \begin{align*}
                    &\Rightarrow \Gamma(y)\,\frac{U_\infty}{2}\,C_\text{A}(y)\,t(y) \\
                    &\Rightarrow C_\text{A}(y) = \frac{2\,\Gamma(y)}{U_\infty\,t(y)} \propto \frac{\Gamma(y)}{t(y)} \\[0.5ex]
                    &\Rightarrow C_\text{A}(\theta) = \frac{2\,\Gamma(\theta)}{U_\infty\,t(\theta)}
                \end{align*}%
                \vspace{1mm}
            \end{minipage}
        }{0}{0}{1}{0}

                \titlerm{Allgemein}\\ \vspace{-4mm}
                \begin{align*}
                    \Gamma(\theta) &= \sum_{n=1}^M A_n\,\sin\!\left(n\,\theta\right) \\
                    \frac{\Gamma(\theta)}{b\,U_\infty} &= \gamma(\theta) = \sum_{n=1}^M a_n\,\sin\!\left(n\,\theta\right) \\
                    \frac{A_n}{b\,U_\infty} &= a_n = \frac{2}{\pi}\,\int_0^\pi \gamma(\theta)\,\sin\!\left(\theta\right)\,\text{d}\theta
                \end{align*}%
                \vspace{-3.2mm}
            \end{minipage}%

                \centering
                \titlerm{Elliptische Zirkulationsverteilung}\\ \vspace{-4mm}
                \begin{align*}
                    \Gamma(y)      &= \Gamma_\text{max}\cdot\sqrt{1-\left(\tfrac{2\,y}{b}\right)^2} \\
                    \Gamma(\eta)   &= \Gamma_\text{max}\cdot\sqrt{1-\eta^2} \\[1.0ex]
                    \Gamma(\theta) &= \Gamma_\text{max}\cdot\sin\!\left(\theta\right)
                \end{align*}
                \titlerm{Tiefenverteilung bei Trapezflügel}\\ \vspace{-4mm}
                \begin{equation*}
                    t(\theta) = t_\text{i} + \left(t_\text{a} - t_\text{i}\right)\,\cos\!\left(\theta\right)
                \end{equation*}

            \end{equation*}

            \end{equation*}

            \begin{equation*}
                f(\eta) = \frac{2\,b}{\left(\frac{\partial C_\text{A}}{\partial\alpha}\right)_\text{2D}\,t(\eta)} \quad\vline\quad \gamma(\eta) = \frac{\Gamma(\eta)}{b\,U_\infty} \quad\vline\quad \eta = \frac{2\,y}{b}
            \end{equation*}
            \rule{\dimexpr\cellwidth-2\TileSep}{0.5pt} \\ \vspace{-4mm}
            \begin{equation*}
                \alpha_\text{g} = \underbrace{f(\theta)\cdot\gamma(\theta)}_{\alpha_\text{e}(\theta)} + \underbrace{\frac{1}{2\,\pi}\,\int_{-1}^1 \frac{\partial\gamma(\theta')}{\partial\theta'}\,\frac{\partial\theta'}{\theta-\theta'}}_{\alpha_\text{i}(\theta)}

            \begin{equation*}
                \cos\!\left(\theta\right) = \eta = \frac{2\,y}{b}
            \end{equation*}

                C_\text{A} = \frac{\pi\,\Lambda\,A_1}{2\,b\,U_\infty} \quad\vline\quad \frac{\text{d}C_\text{A}}{\text{d}\alpha} = \frac{2\,\pi\,\Lambda}{\Lambda + 2} \quad\vline\quad \Lambda = \frac{b^2}{S}
            \end{equation*}
            \begin{equation*}
                C_\text{W\!,i} = \frac{C_\text{A}^2}{\pi\,\Lambda}\,\underbrace{\left(1 + \frac{1}{A_1^2}\,\sum_{n=2}^M n\,A_n^2\right)}_{k} = k\cdot C_\text{W\!,i\,ell.}

                C_\text{A,\,ell.}  &= \frac{\pi\,b\,\Gamma_\text{max}}{2\,U_\infty\,F} = \frac{\pi\,\Lambda\,\Gamma_\text{max}}{2\,b\,U_\infty} \\[0.5ex]
                C_\text{W\!,i\,ell.} &= \frac{C_\text{A}^2}{\pi\,\Lambda}
            \end{align*}%
            \vspace{-1mm} \\
            \titlerm{Wenn ungeschränkt $\Gamma(y) \propto t(y)$}\\ \vspace{-4mm}
            \begin{align*}
                C_\text{A,\,ell.}  &= \frac{\left(\frac{\partial C_\text{A}}{\partial\alpha}\right)_\text{2D}\,\pi\,\Lambda}{\pi\,\Lambda + \left(\frac{\partial C_\text{A}}{\partial\alpha}\right)_\text{2D}}\cdot\alpha_\text{g} \\[0.5ex]
                \left(\frac{\partial C_\text{A}}{\partial\alpha}\right)_\text{2D} &= 2\,\pi
            \end{align*}
        }{0}{0}{1}{0}

                                &= \frac{1}{\sin\!\left(\theta\right)}\,\sum_{n=1}^M \frac{A_n}{b\,U_\infty}\,\sin\!\left(n\,\theta\right)\left[\frac{4\,b\,\sin\!\left(\theta\right)}{\left(\frac{\partial C_\text{A}}{\partial\alpha}\right)_\text{2D}\,t(\theta)} + n\right]
